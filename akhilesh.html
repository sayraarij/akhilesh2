        // Create floating hearts

        function createHeart() {

            const heart = document.createElement('div');

            heart.className = 'heart';

            heart.textContent = '❤️';

            heart.style.left = Math.random() * 100 + 'vw';

            heart.style.top = Math.random() * 100 + 'vh';

            heart.style.animationDelay = Math.random() * 3 + 's';

            heart.style.animationDuration = (Math.random() * 3 + 4) + 's';

            document.body.appendChild(heart);

        }


        for (let i = 0; i < 15; i++) {

            createHeart();

        }


        const noBtn = document.getElementById('noBtn');

        const yesBtn = document.getElementById('yesBtn');

        const question = document.getElementById('question');

        const successMsg = document.getElementById('successMsg');

        const buttonsDiv = document.querySelector('.buttons');


        // Make No button run away but stay in view

        noBtn.addEventListener('mouseenter', () => {

            const containerRect = buttonsDiv.getBoundingClientRect();

            const btnRect = noBtn.getBoundingClientRect();

            

            // Random position within the button container area

            const maxX = containerRect.width - btnRect.width;

            const x = Math.random() * maxX;

            

            noBtn.style.transform = `translateX(${x - btnRect.left + containerRect.left}px)`;

            setTimeout(() => {

                noBtn.style.transform = 'translateX(0)';

            }, 500);

        });


        // For mobile touch

        noBtn.addEventListener('touchstart', (e) => {

            e.preventDefault();

            const containerRect = buttonsDiv.getBoundingClientRect();

            const btnRect = noBtn.getBoundingClientRect();

            

            const maxX = containerRect.width - btnRect.width;

            const x = Math.random() * maxX;

            

            noBtn.style.transform = `translateX(${x - btnRect.left + containerRect.left}px)`;

            setTimeout(() => {

                noBtn.style.transform = 'translateX(0)';

            }, 500);

        });
